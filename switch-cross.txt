# Meson cross file for Nintendo Switch homebrew (devkitA64 / libnx / devkitPro)
#
# Usage (run through switch-env.sh so TMP/DEVKITPRO are in Windows form):
#   bash switch-env.sh bash -c 'meson setup build-switch --cross-file=switch-cross.txt --wipe \
#         -Dgfx_backend=gles -Ddefault_system=enabled \
#         -Dworkdir_current=true -Duse_miniffi=false'
#
# Requires: DEVKITPRO=C:/devkitPro with devkitA64, libnx, switch-pkg-config and
#           portlibs under C:/devkitPro/portlibs/switch (SDL2, mesa/EGL, ...).
#
# NOTE: meson's machine-file parser does not accept multi-line arrays; every
# array below is deliberately written on a single line. Do not reflow them.
#
# NOTE (pkg-config on this host, verified):
#   * msys2 pkgconf splits PKG_CONFIG_PATH on ':' only. Meson joins multiple
#     pkg_config_path entries with ';' on Windows, which the binary cannot
#     parse, so the list below MUST stay a single directory. Everything
#     (SDL2, mesa/EGL, OpenAL, PhysFS, vorbis, ...) has to land in
#     portlibs/switch/lib/pkgconfig, which is the case for devkitPro portlibs.
#   * the paths must be POSIX form (/c/devkitPro/...). Drive-letter forward
#     slash form ('C:/devkitPro/...') is not resolved by this pkgconf build.
#   * devkitPro .pc files carry prefix=/opt/devkitpro/..., which native
#     aarch64-none-elf-gcc cannot open (verified: "zlib.h: No such file").
#     Either rewrite them at install time (sed s#/opt/devkitpro#C:/devkitPro#)
#     or rely on the explicit -I/-L in [built-in options] below, which cover
#     exactly those two directories.

[binaries]
c        = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-gcc.exe'
cpp      = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-g++.exe'
ar       = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-ar.exe'
strip    = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-strip.exe'
ld       = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-ld.exe'
nm       = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-nm.exe'
objcopy  = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-objcopy.exe'
objdump  = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-objdump.exe'
ranlib   = 'C:/devkitPro/devkitA64/bin/aarch64-none-elf-ranlib.exe'
# switch-pkg-config ships a bash wrapper (aarch64-none-elf-pkg-config); Windows
# cannot exec a shebang script through CreateProcess, so use the native
# pkg-config and constrain the search path with pkg_config_path below.
pkg-config = 'C:/devkitPro/msys2/usr/bin/pkg-config.exe'

[properties]
# No Switch emulator is wired up, so meson must not try to run target binaries.
needs_exe_wrapper = true
sys_root = 'C:/devkitPro/devkitA64/aarch64-none-elf'

[host_machine]
system     = 'switch'
cpu_family = 'aarch64'
cpu        = 'cortex-a57'
endian     = 'little'

[built-in options]
c_args = ['-D__SWITCH__', '-march=armv8-a+crc+crypto', '-mtune=cortex-a57', '-mtp=soft', '-fPIE', '-ffunction-sections', '-fdata-sections', '-IC:/devkitPro/devkitA64/aarch64-none-elf/include', '-IC:/devkitPro/portlibs/switch/include', '-IC:/devkitPro/libnx/include']
cpp_args = ['-D__SWITCH__', '-march=armv8-a+crc+crypto', '-mtune=cortex-a57', '-mtp=soft', '-fPIE', '-ffunction-sections', '-fdata-sections', '-IC:/devkitPro/devkitA64/aarch64-none-elf/include', '-IC:/devkitPro/portlibs/switch/include', '-IC:/devkitPro/libnx/include']
# devkitPro passes ARCH (-fPIE) to the link step too; without it ld links as a
# non-PIE executable and switch.ld fails with "read-only segment has dynamic
# relocations".
c_link_args = ['-fPIE', '-specs=C:/devkitPro/libnx/switch.specs', '-march=armv8-a+crc+crypto', '-mtune=cortex-a57', '-mtp=soft', '-Wl,--gc-sections', '-LC:/devkitPro/portlibs/switch/lib', '-LC:/devkitPro/libnx/lib', '-lnx']
cpp_link_args = ['-fPIE', '-specs=C:/devkitPro/libnx/switch.specs', '-march=armv8-a+crc+crypto', '-mtune=cortex-a57', '-mtp=soft', '-Wl,--gc-sections', '-LC:/devkitPro/portlibs/switch/lib', '-LC:/devkitPro/libnx/lib', '-lnx']
pkg_config_path = ['/c/devkitPro/portlibs/switch/lib/pkgconfig']
